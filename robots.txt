# --- AI crawlers -----------------------------------------------------------
# Policy: indexable for SEARCH only. No model training, no use as AI input /
# RAG grounding context. Declared machine-readably below (Cloudflare Content
# Signals), in /ai.txt, and via <meta name="tdm-reservation"> on every page.
#
# Content-Signal: search=yes, ai-input=no, ai-train=no
#   search   = show this page as a link/snippet in a result list  -> allowed
#   ai-input = feed this page to a model to generate an answer    -> NOT allowed
#   ai-train = use this page as training/fine-tuning corpus       -> NOT allowed
Content-Signal: search=yes, ai-input=no, ai-train=no
